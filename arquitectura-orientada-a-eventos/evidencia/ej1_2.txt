# Ejercicio 1: un consumidor recibe
$ ./eda api-server        # terminal: api
$ ./eda consumer --topic orderCreated        # terminal: c1
$ curl -s -X POST localhost:3000/topics/orderCreated -H 'Content-Type: application/json' -d '{"order_id":123,"total":49.99}'
{"message":"message published"}
--- salida de la terminal c1 ---
